local function Moon(Star)
    local Cloud = {
        a="ᴀ",b="ʙ",c="ᴄ",d="ᴅ",e="ᴇ",f="ꜰ",g="ɢ",h="ʜ",i="ɪ",j="ᴊ",k="ᴋ",l="ʟ",m="ᴍ",
        n="ɴ",o="ᴏ",p="ᴘ",q="ǫ",r="ʀ",s="s",t="ᴛ",u="ᴜ",v="ᴠ",w="ᴡ",x="x",y="ʏ",z="ᴢ",
        A="ᴀ",B="ʙ",C="ᴄ",D="ᴅ",E="ᴇ",F="ꜰ",G="ɢ",H="ʜ",I="ɪ",J="ᴊ",K="ᴋ",L="ʟ",M="ᴍ",
        N="ɴ",O="ᴏ",P="ᴘ",Q="ǫ",R="ʀ",S="s",T="ᴛ",U="ᴜ",V="ᴠ",W="ᴡ",X="x",Y="ʏ",Z="ᴢ"
    }
    local Sky = ""
    for Wind = 1, #Star do
        local Rain = Star:sub(Wind, Wind)
        Sky = Sky .. (Cloud[Rain] or Rain)
    end
    return Sky
end

local function convertRGB(value)
    local RGBcolour = { ["0"]="9", ["1"]="0", ["2"]="1", ["3"]="2", ["4"]="3", ["5"]="4", ["6"]="5", ["7"]="6", ["8"]="7", ["9"]="8" }
    local result = {}
    for i = 1, #value do
        local digit = value:sub(i, i)
        result[i] = RGBcolour[digit] or digit
    end
    return table.concat(result)
end

-- Volume Booster
local MAX_VOLUME = 5
local function maxVolume(obj)
    if not obj then return end
    if obj:IsA("Sound") then
        obj.Volume = MAX_VOLUME
        return
    end
    for _, v in ipairs(obj:GetDescendants()) do
        if v:IsA("Sound") then v.Volume = MAX_VOLUME end
    end
end

local function AddButtonEffects(btn, defaultColor, hoverColor)
    local scale = Instance.new("UIScale", btn)
    btn.MouseEnter:Connect(function()
        TweenService:Create(btn, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundColor3 = hoverColor}):Play()
        TweenService:Create(scale, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1.03}):Play()

local isToggledOn = false
local function attemptVehicleSpawn()
    mainRE:FireServer("SegwaySmall")
    local vehicleSpawned = false
    local connections = {}
    local function onChildAdded(child)
        if child:IsA("Model") or child:IsA("Tool") then vehicleSpawned = true end
    end
    table.insert(connections, workspace.ChildAdded:Connect(onChildAdded))
    if workspace:FindFirstChild("Vehicles") then table.insert(connections, workspace.Vehicles.ChildAdded:Connect(onChildAdded)) end
    if player.Character then table.insert(connections, player.Character.ChildAdded:Connect(onChildAdded)) end
    
    task.wait(1.5)
    for _, conn in ipairs(connections) do conn:Disconnect() end
    
    if isToggledOn and not vehicleSpawned then mainRE:FireServer("SkateBoard") end
    pcall(function()
        local volEvent = game:GetService("Players").LocalPlayer.PlayerGui.MainGUIHandler.MainAudio.Catalog.Header.VolumeContainer.VolumeChangeRequest
        volEvent:FireServer(100)
    end)
end
